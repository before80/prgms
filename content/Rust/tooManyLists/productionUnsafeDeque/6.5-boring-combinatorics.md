好，回到我们的常规链表节目！

先把 `Drop` 搞定，用 pop 就行：

```rust
impl<T> Drop for LinkedList<T> {
    fn drop(&mut self) {
        // 一直 pop 直到不得不停
        while let Some(_) = self.pop_front() { }
    }
}
```

还得填一堆很无聊的组合实现：`front`、`front_mut`、`back`、`back_mut`、`iter`、`iter_mut`、`into_iter`……

可以用宏什么的，但老实说，那比复制粘贴还惨。我们就大量复制粘贴。我*非常仔细*地设计了前面的 push/pop，这样应该能*字面意义上*只 swap front 和 back，代码就对上了！惨痛经验万岁！（很想用 prev/next 说节点，但我觉得尽量统一说 front/back 值得，能少犯错。）

好，先 `front`：

```rust
pub fn front(&self) -> Option<&T> {
    unsafe {
        self.front.map(|node| &(*node.as_ptr()).elem)
    }
}
```

嘿，这书很老了，有些 nice 新东西比如 `?` 运算符，在 Option::None 时 early return，能让代码更好吗？

```rust
pub fn front(&self) -> Option<&T> {
    unsafe {
        Some(&(*self.front?.as_ptr()).elem)
    }
}
```

也许？这么简单的东西差不多，上一节刚说提前返回对我们有点吓人，也许这里更明确更好（我坚持用 `map` 实现）。继续 front_mut：

```rust
pub fn front_mut(&mut self) -> Option<&mut T> {
    unsafe {
        self.front.map(|node| &mut (*node.as_ptr()).elem)
    }
}
```

`back` 版本我丢文末。

接下来迭代器。和之前所有链表不同，我们*终于*能搞 [DoubleEndedIterator](https://doc.rust-lang.org/std/iter/trait.DoubleEndedIterator.html)，要 production quality 还得 [ExactSizeIterator](https://doc.rust-lang.org/std/iter/trait.ExactSizeIterator.html)。

所以除了 `next` 和 `size_hint`，还要 `next_back` 和 `len`。

眼尖的会发现 IterMut 做双端迭代更 sketchy，但其实仍然 sound！

…… god 这要大量 boilerplate。也许该写宏……不，那还是更惨。

```rust
pub struct Iter<'a, T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<&'a T>,
}

impl<T> LinkedList<T> {
    pub fn iter(&self) -> Iter<T> {
        Iter { 
            front: self.front, 
            back: self.back,
            len: self.len,
            _boo: PhantomData,
        }
    }
}


impl<'a, T> IntoIterator for &'a LinkedList<T> {
    type IntoIter = Iter<'a, T>;
    type Item = &'a T;

    fn into_iter(self) -> Self::IntoIter {
        self.iter()
    }
}

impl<'a, T> Iterator for Iter<'a, T> {
    type Item = &'a T;
    
    fn next(&mut self) -> Option<Self::Item> {
        // 这里检查 self.front == self.back 很诱人，
        // 但对产出最后一个元素不对！那种
        // 做法只对数组有效，因为有「越界一端」指针。
        if self.len > 0 {
            // 可以 unwrap front，但这样更安全简单
            self.front.map(|node| unsafe {
                self.len -= 1;
                self.front = (*node.as_ptr()).back;
                &(*node.as_ptr()).elem
            })
        } else {
            None
        }
    }

    fn size_hint(&self) -> (usize, Option<usize>) {
        (self.len, Some(self.len))
    }
}

impl<'a, T> DoubleEndedIterator for Iter<'a, T> {
    fn next_back(&mut self) -> Option<Self::Item> {
        if self.len > 0 {
            self.back.map(|node| unsafe {
                self.len -= 1;
                self.back = (*node.as_ptr()).front;
                &(*node.as_ptr()).elem
            })
        } else {
            None
        }
    }
}

impl<'a, T> ExactSizeIterator for Iter<'a, T> {
    fn len(&self) -> usize {
        self.len
    }
}
```

……那只是 `.iter()`……

IterMut 贴文末，literally 一样只是很多地方加 `mut`，先搞定 `into_iter`。还能 mercifully 用老办法：包一层 collection，用 pop 当 next：

```rust
pub struct IntoIter<T> {
    list: LinkedList<T>,
}

impl<T> LinkedList<T> {
    pub fn into_iter(self) -> IntoIter<T> {

        }
    }
}

        self.into_iter()
    }
}

        self.list.pop_front()

        self.list.pop_back()
    }
}

    fn len(&self) -> usize {
        self.list.len
    }
}
```

还是一坨 boilerplate，至少是*satisfying* boilerplate。

好，组合都填完的全部代码：

```rust
use std::ptr::NonNull;
use std::marker::PhantomData;

pub struct LinkedList<T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<T>,
}

type Link<T> = Option<NonNull<Node<T>>>;

struct Node<T> {
    front: Link<T>,
    back: Link<T>,
    elem: T, 
}

pub struct Iter<'a, T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<&'a T>,
}

pub struct IterMut<'a, T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<&'a mut T>,
}

pub struct IntoIter<T> {
    list: LinkedList<T>,
}

impl<T> LinkedList<T> {
    pub fn new() -> Self {
        Self {
            front: None,
            back: None,
            len: 0,
            _boo: PhantomData,
        }
    }

    pub fn push_front(&mut self, elem: T) {
        // SAFETY: 这是链表，你还想怎样？
        unsafe {
            let new = NonNull::new_unchecked(Box::into_raw(Box::new(Node {
                front: None,
                back: None,
                elem,
            })));
            if let Some(old) = self.front {
                // 把新 front 插到旧的前面
                (*old.as_ptr()).front = Some(new);
                (*new.as_ptr()).back = Some(old);
            } else {
                // 如果没有 front，说明是空链表，也要设置 back。
                self.back = Some(new);
            }
            // 这些总会发生！
            self.front = Some(new);
            self.len += 1;
        }
    }

    pub fn push_back(&mut self, elem: T) {
        // SAFETY: 这是链表，你还想怎样？
        unsafe {
            let new = NonNull::new_unchecked(Box::into_raw(Box::new(Node {
                back: None,
                front: None,
                elem,
            })));
            if let Some(old) = self.back {
                // 把新 back 插到旧的后面
                (*old.as_ptr()).back = Some(new);
                (*new.as_ptr()).front = Some(old);
            } else {
                // 如果没有 back，说明是空链表，也要设置 front。
                self.front = Some(new);
            }
            // 这些总会发生！
            self.back = Some(new);
            self.len += 1;
        }
    }

    pub fn pop_front(&mut self) -> Option<T> {
        unsafe {
            // 只有存在 front 节点要 pop 时才做事。
            self.front.map(|node| {
                // 让 Box 复活，以便移出值并 Drop 它
                // （Box 继续神奇地帮我们处理）。
                let boxed_node = Box::from_raw(node.as_ptr());
                let result = boxed_node.elem;

                // 让下一个节点成为新 front。
                self.front = boxed_node.back;
                if let Some(new) = self.front {
                    // 清理它对已移除节点的引用
                    (*new.as_ptr()).front = None;
                } else {
                    // front 现在是 null，链表空了！
                    self.back = None;
                }

                self.len -= 1;
                result
                // Box 在这里隐式释放，知道里面没有 T。
            })
        }
    }

    pub fn pop_back(&mut self) -> Option<T> {
        unsafe {
            // 只有存在 back 节点要 pop 时才做事。
            self.back.map(|node| {
                // 让 Box 复活，以便移出值并 Drop 它
                // （Box 继续神奇地帮我们处理）。
                let boxed_node = Box::from_raw(node.as_ptr());
                let result = boxed_node.elem;

                // 让下一个节点成为新 back。
                self.back = boxed_node.front;
                if let Some(new) = self.back {
                    // 清理它对已移除节点的引用
                    (*new.as_ptr()).back = None;
                } else {
                    // back 现在是 null，链表空了！
                    self.front = None;
                }

                self.len -= 1;
                result
                // Box 在这里隐式释放，知道里面没有 T。
            })
        }
    }

    pub fn front(&self) -> Option<&T> {
        unsafe {
            self.front.map(|node| &(*node.as_ptr()).elem)
        }
    }

    pub fn front_mut(&mut self) -> Option<&mut T> {
        unsafe {
            self.front.map(|node| &mut (*node.as_ptr()).elem)
        }
    }

    pub fn back(&self) -> Option<&T> {
        unsafe {
            self.back.map(|node| &(*node.as_ptr()).elem)
        }
    }

    pub fn back_mut(&mut self) -> Option<&mut T> {
        unsafe {
            self.back.map(|node| &mut (*node.as_ptr()).elem)
        }
    }

    pub fn len(&self) -> usize {
        self.len
    }

    pub fn iter(&self) -> Iter<T> {
        Iter { 
            front: self.front, 
            back: self.back,
            len: self.len,
            _boo: PhantomData,
        }
    }

    pub fn iter_mut(&mut self) -> IterMut<T> {
        IterMut { 
            front: self.front, 
            back: self.back,
            len: self.len,
            _boo: PhantomData,
        }
    }

    pub fn into_iter(self) -> IntoIter<T> {
        IntoIter { 
            list: self
        }
    }
}

impl<T> Drop for LinkedList<T> {
    fn drop(&mut self) {
        // 一直 pop 直到不得不停
        while let Some(_) = self.pop_front() { }
    }
}

    type IntoIter = Iter<'a, T>;
    type Item = &'a T;

    fn into_iter(self) -> Self::IntoIter {
        self.iter()
    }
}

    type Item = &'a T;

    fn next(&mut self) -> Option<Self::Item> {

        // 但对产出最后一个元素不对！那种
        // 做法只对数组有效，因为有「越界一端」指针。
        if self.len > 0 {
            // 可以 unwrap front，但这样更安全简单
            self.front.map(|node| unsafe {
                self.len -= 1;
                self.front = (*node.as_ptr()).back;
                &(*node.as_ptr()).elem
            })
        } else {
            None
        }
    }

    fn size_hint(&self) -> (usize, Option<usize>) {
        (self.len, Some(self.len))
    }
}

impl<'a, T> DoubleEndedIterator for Iter<'a, T> {
    fn next_back(&mut self) -> Option<Self::Item> {
        if self.len > 0 {
            self.back.map(|node| unsafe {
                self.len -= 1;
                self.back = (*node.as_ptr()).front;

        }
    }
}

impl<'a, T> ExactSizeIterator for Iter<'a, T> {
    fn len(&self) -> usize {
        self.len
    }
}

    type IntoIter = IterMut<'a, T>;

        self.into_iter()

}
```
