```rust
pub fn peek_next(&mut self) -> Option<&mut T> {
    unsafe {
        self.cur
            .and_then(|node| (*node.as_ptr()).back)
            .map(|node| &mut (*node.as_ptr()).elem)
    }
}
```

……对，就是错。`self.cur` 是 None 不能 give up，还要查 `self.list.front`，因为在 ghost 上！链上加 or_else：

```rust
pub fn peek_next(&mut self) -> Option<&mut T> {
    unsafe {
        self.cur
            .and_then(|node| (*node.as_ptr()).back)
            .or_else(|| self.list.front)
            .map(|node| &mut (*node.as_ptr()).elem)
    }
}

pub fn peek_prev(&mut self) -> Option<&mut T> {
    unsafe {
        self.cur
            .and_then(|node| (*node.as_ptr()).front)
            .or_else(|| self.list.back)
            .map(|node| &mut (*node.as_ptr()).elem)
    }
}
```

```rust
pub fn peek_next(&mut self) -> Option<&mut T> {
    unsafe {
        let next = if let Some(cur) = self.cur {
            // 正常情况，跟随 cur 节点的 back 指针
            (*cur.as_ptr()).back
        } else {
            // ghost 情况，用链表的 front 指针
            self.list.front
        };

        // 若 next 节点存在则产出元素
        next.map(|node| &mut (*node.as_ptr()).elem)
    }
}

pub fn peek_prev(&mut self) -> Option<&mut T> {
    unsafe {
        let prev = if let Some(cur) = self.cur {
            // 正常情况，跟随 cur 节点的 front 指针
            (*cur.as_ptr()).front
        } else {
            // ghost 情况，用链表的 back 指针
            self.list.back
        };

        // 若 prev 节点存在则产出元素
        prev.map(|node| &mut (*node.as_ptr()).elem)
    }
}
```
