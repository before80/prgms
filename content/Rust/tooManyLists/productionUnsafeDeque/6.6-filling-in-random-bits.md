这些我肯定全是 scratch 写的，才不是 copy std impls。因为它们 so interesting，我 definitely 记得手动实现 Hash 的 subtlety。对，这是我 All The Time 在想的事……

其实有几处值得说。

第一，讨厌的命名空间冲突。不知为何 std 现在有叫 Hash 和 Debug 的宏，没 import trait 时会得到关于宏的 cryptic 错误，而不是正确的「missing trait」。

另一个有趣的是 Hash 本身。看到我们对 `len` 做 hash 了吗？这 really important！集合不对长度 hash，[可能意外让自己 vulnerable to prefix collisions](https://doc.rust-lang.org/std/hash/trait.Hash.html#prefix-collisions)。比如什么区分 `["he", "llo"]` 和 `["hello"]`？若没人 hash 长度或其他「分隔符」，nothing！让 hash collision 太容易意外或恶意发生会导致 serious sadness，所以 just do it！

好，当前代码：

```rust
use std::cmp::Ordering;
use std::fmt::{self, Debug};
use std::hash::{Hash, Hasher};
use std::iter::FromIterator;
use std::ptr::NonNull;
use std::marker::PhantomData;

pub struct LinkedList<T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<T>,
}

type Link<T> = Option<NonNull<Node<T>>>;

struct Node<T> {
    front: Link<T>,
    back: Link<T>,
    elem: T, 
}

pub struct Iter<'a, T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<&'a T>,
}

pub struct IterMut<'a, T> {
    front: Link<T>,
    back: Link<T>,
    len: usize,
    _boo: PhantomData<&'a mut T>,
}

pub struct IntoIter<T> {
    list: LinkedList<T>,
}

impl<T> LinkedList<T> {
    pub fn new() -> Self {
        Self {
            front: None,
            back: None,
            len: 0,
            _boo: PhantomData,
        }
    }

    pub fn push_front(&mut self, elem: T) {
        // SAFETY: 这是链表，你还想怎样？
        unsafe {
            let new = NonNull::new_unchecked(Box::into_raw(Box::new(Node {
                front: None,
                back: None,
                elem,
            })));
            if let Some(old) = self.front {
                // 把新 front 插到旧的前面
                (*old.as_ptr()).front = Some(new);
                (*new.as_ptr()).back = Some(old);
            } else {
                // 如果没有 front，说明是空链表，也要设置 back。
                self.back = Some(new);
            }
            // 这些总会发生！
            self.front = Some(new);
            self.len += 1;
        }
    }

    pub fn push_back(&mut self, elem: T) {
        // SAFETY: 这是链表，你还想怎样？
        unsafe {
            let new = NonNull::new_unchecked(Box::into_raw(Box::new(Node {
                back: None,
                front: None,
                elem,
            })));
            if let Some(old) = self.back {
                // 把新 back 插到旧的后面
                (*old.as_ptr()).back = Some(new);
                (*new.as_ptr()).front = Some(old);
            } else {
                // 如果没有 back，说明是空链表，也要设置 front。
                self.front = Some(new);
            }
            // 这些总会发生！
            self.back = Some(new);
            self.len += 1;
        }
    }

    pub fn pop_front(&mut self) -> Option<T> {
        unsafe {
            // 只有存在 front 节点要 pop 时才做事。
            self.front.map(|node| {
                // 让 Box 复活，以便移出值并 Drop 它
                // （Box 继续神奇地帮我们处理）。
                let boxed_node = Box::from_raw(node.as_ptr());
                let result = boxed_node.elem;

                // 让下一个节点成为新 front。
                self.front = boxed_node.back;
                if let Some(new) = self.front {
                    // 清理它对已移除节点的引用
                    (*new.as_ptr()).front = None;
                } else {
                    // front 现在是 null，链表空了！
                    self.back = None;
                }

                self.len -= 1;
                result
                // Box 在这里隐式释放，知道里面没有 T。
            })
        }
    }

    pub fn pop_back(&mut self) -> Option<T> {
        unsafe {
            // 只有存在 back 节点要 pop 时才做事。
            self.back.map(|node| {
                // 让 Box 复活，以便移出值并 Drop 它
                // （Box 继续神奇地帮我们处理）。
                let boxed_node = Box::from_raw(node.as_ptr());
                let result = boxed_node.elem;

                // 让下一个节点成为新 back。
                self.back = boxed_node.front;
                if let Some(new) = self.back {
                    // 清理它对已移除节点的引用
                    (*new.as_ptr()).back = None;
                } else {
                    // back 现在是 null，链表空了！
                    self.front = None;
                }

                self.len -= 1;
                result
                // Box 在这里隐式释放，知道里面没有 T。
            })
        }
    }

    pub fn front(&self) -> Option<&T> {
        unsafe {
            self.front.map(|node| &(*node.as_ptr()).elem)
        }
    }

    pub fn front_mut(&mut self) -> Option<&mut T> {
        unsafe {
            self.front.map(|node| &mut (*node.as_ptr()).elem)
        }
    }

    pub fn back(&self) -> Option<&T> {
        unsafe {
            self.back.map(|node| &(*node.as_ptr()).elem)
        }
    }

    pub fn back_mut(&mut self) -> Option<&mut T> {
        unsafe {
            self.back.map(|node| &mut (*node.as_ptr()).elem)
        }
    }

    pub fn len(&self) -> usize {
        self.len
    }

    pub fn is_empty(&self) -> bool {
        self.len == 0
    }

    pub fn clear(&mut self) {
        // 看，又是 drop
        while let Some(_) = self.pop_front() { }
    }

    pub fn iter(&self) -> Iter<T> {
        Iter { 
            front: self.front, 
            back: self.back,
            len: self.len,
            _boo: PhantomData,
        }
    }

    pub fn iter_mut(&mut self) -> IterMut<T> {
        IterMut { 
            front: self.front, 
            back: self.back,
            len: self.len,
            _boo: PhantomData,
        }
    }

    pub fn into_iter(self) -> IntoIter<T> {
        IntoIter { 
            list: self
        }
    }
}

impl<'a, T> IntoIterator for &'a LinkedList<T> {
    type IntoIter = Iter<'a, T>;
    type Item = &'a T;

    fn into_iter(self) -> Self::IntoIter {
        self.iter()
    }
}

impl<'a, T> Iterator for Iter<'a, T> {
    type Item = &'a T;

    fn next(&mut self) -> Option<Self::Item> {
        // 这里检查 self.front == self.back 很诱人，
        // 但对产出最后一个元素不对！那种
        // 做法只对数组有效，因为有「越界一端」指针。
        if self.len > 0 {
            // 可以 unwrap front，但这样更安全简单
            self.front.map(|node| unsafe {
                self.len -= 1;
                self.front = (*node.as_ptr()).back;
                &(*node.as_ptr()).elem
            })
        } else {
            None
        }
    }

    fn size_hint(&self) -> (usize, Option<usize>) {
        (self.len, Some(self.len))
    }
}

impl<'a, T> DoubleEndedIterator for Iter<'a, T> {
    fn next_back(&mut self) -> Option<Self::Item> {
        if self.len > 0 {
            self.back.map(|node| unsafe {
                self.len -= 1;
                self.back = (*node.as_ptr()).front;
                &(*node.as_ptr()).elem
            })
        } else {
            None
        }
    }
}

impl<'a, T> ExactSizeIterator for Iter<'a, T> {
    fn len(&self) -> usize {
        self.len
    }
}
```
